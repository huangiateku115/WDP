<script>
  (function() {
    const tabs = document.querySelectorAll('.filter-tab');
    const items = document.querySelectorAll('.notification-item');
    const emptyState = document.getElementById('emptyNoticeState');
    const markAllBtn = document.getElementById('markAllReadBtn');
    const groups = document.querySelectorAll('.notification-group');

    // Filter Logic
    tabs.forEach(tab => {
      tab.addEventListener('click', () => {
        // Toggle active styling
        tabs.forEach(t => {
          t.classList.remove('bg-primary', 'text-on-primary');
          t.classList.add('bg-surface-container-high', 'text-on-surface-variant');
        });
        tab.classList.remove('bg-surface-container-high', 'text-on-surface-variant');
        tab.classList.add('bg-primary', 'text-on-primary');

        const filter = tab.getAttribute('data-filter');
        let visibleCount = 0;

        items.forEach(item => {
          const type = item.getAttribute('data-type');
          const isRead = item.getAttribute('data-read') === 'true';

          let show = false;
          if (filter === 'all') {
            show = true;
          } else if (filter === 'unread') {
            show = !isRead;
          } else if (filter === 'contract') {
            show = type === 'contract';
          } else if (filter === 'inventory') {
            show = type === 'inventory';
          }

          if (show) {
            item.classList.remove('hidden');
            visibleCount++;
          } else {
            item.classList.add('hidden');
          }
        });

        // Hide empty group headers
        groups.forEach(g => {
          const groupVisibleItems = g.querySelectorAll('.notification-item:not(.hidden)');
          if (groupVisibleItems.length === 0) {
            g.classList.add('hidden');
          } else {
            g.classList.remove('hidden');
          }
        });

        // Handle overall empty state
        if (visibleCount === 0) {
          emptyState.classList.remove('hidden');
          emptyState.classList.add('flex');
        } else {
          emptyState.classList.add('hidden');
          emptyState.classList.remove('flex');
        }
      });
    });

    // Mark all as read micro-interaction
    markAllBtn.addEventListener('click', () => {
      items.forEach(item => {
        item.setAttribute('data-read', 'true');
        item.classList.remove('bg-surface-container-lowest', 'shadow-sm');
        item.classList.add('bg-surface-container-low');
        
        const dot = item.querySelector('.unread-indicator');
        if (dot) {
          dot.remove();
        }
      });

      // Update unread tab counter badge
      const unreadTabBadge = document.querySelector('[data-filter="unread"] span:last-child');
      if (unreadTabBadge) {
        unreadTabBadge.textContent = '0';
        unreadTabBadge.classList.remove('bg-secondary');
        unreadTabBadge.classList.add('bg-surface-variant', 'text-on-surface-variant');
      }

      // Quick visual feedback
      markAllBtn.innerHTML = `
        <span class="material-symbols-outlined text-[16px] text-primary">check_circle</span>
        <span class="text-on-surface-variant">Đã đánh dấu xong</span>
      `;
      markAllBtn.disabled = true;
    });

    // Single item click to mark as read
    items.forEach(item => {
      item.addEventListener('click', () => {
        if (item.getAttribute('data-read') === 'false') {
          item.setAttribute('data-read', 'true');
          item.classList.remove('bg-surface-container-lowest', 'shadow-sm');
          item.classList.add('bg-surface-container-low');
          const dot = item.querySelector('.unread-indicator');
          if (dot) dot.remove();

          // Decrement badge if > 0
          const unreadTabBadge = document.querySelector('[data-filter="unread"] span:last-child');
          if (unreadTabBadge) {
            let current = parseInt(unreadTabBadge.textContent) || 0;
            if (current > 0) {
              current -= 1;
              unreadTabBadge.textContent = current;
              if (current === 0) {
                unreadTabBadge.classList.remove('bg-secondary');
                unreadTabBadge.classList.add('bg-surface-variant', 'text-on-surface-variant');
              }
            }
          }
        }
      });
    });
  })();

  function filterTimeline(type) {
    const items = document.querySelectorAll('.timeline-item');
    const pills = document.querySelectorAll('.filter-pill');
    let visibleCount = 0;

    // Reset pill styles
    pills.forEach(pill => {
      pill.classList.remove('bg-primary', 'text-on-primary', 'shadow-sm');
      pill.classList.add('bg-surface-container', 'text-on-surface-variant');
    });

    // Activate current pill
    const activeBtn = document.getElementById('btn-filter-' + type);
    if (activeBtn) {
      activeBtn.classList.remove('bg-surface-container', 'text-on-surface-variant');
      activeBtn.classList.add('bg-primary', 'text-on-primary', 'shadow-sm');
    }

    // Toggle items
    items.forEach(item => {
      if (type === 'all' || item.getAttribute('data-category') === type) {
        item.style.display = 'flex';
        visibleCount++;
      } else {
        item.style.display = 'none';
      }
    });

    // Empty state trigger
    const emptyState = document.getElementById('timelineEmptyState');
    if (emptyState) {
      if (visibleCount === 0) {
        emptyState.classList.remove('hidden');
      } else {
        emptyState.classList.add('hidden');
      }
    }
  }

<script>
  function calculateWeightAndVolume() {
    const qty = parseFloat(document.getElementById('quantityInput').value) || 0;
    const unitWt = parseFloat(document.getElementById('unitWeightInput').value) || 0;
    const l = parseFloat(document.getElementById('lenInput').value) || 0;
    const w = parseFloat(document.getElementById('widInput').value) || 0;
    const h = parseFloat(document.getElementById('heiInput').value) || 0;

    const totalWeight = Math.round(qty * unitWt);
    document.getElementById('totalWeightDisplay').textContent = totalWeight + ' kg';

    const unitVolume = l * w * h;
    const totalVolume = (unitVolume * qty).toFixed(2);
    document.getElementById('volumePreview').textContent = '~ ' + totalVolume + ' m³';
  }

  function generateRandomSKU() {
    const randNum = Math.floor(1000 + Math.random() * 9000);
    const skuField = document.getElementById('skuInput');
    skuField.value = 'SKU-CF-' + randNum;
    showToast('Đã cấp mã tự động: ' + skuField.value);
  }

  function copySKU() {
    const sku = document.getElementById('skuInput').value;
    navigator.clipboard?.writeText(sku);
    showToast('Đã sao chép mã ' + sku + ' vào bộ nhớ tạm');
  }

  function showToast(msg) {
    const toast = document.getElementById('toastMessage');
    toast.querySelector('span.text-body-md-medium').textContent = msg;
    toast.classList.remove('hidden');
    toast.classList.add('flex');
    setTimeout(() => {
      toast.classList.add('hidden');
      toast.classList.remove('flex');
    }, 3000);
  }

  function updatePhotoCount() {
    const photos = document.querySelectorAll('#photoGallery > div').length;
    document.getElementById('photoCount').textContent = photos + '/10 ảnh';
  }

  function mockAddPhoto() {
    const gallery = document.getElementById('photoGallery');
    const existing = gallery.querySelectorAll('div').length;
    if (existing >= 10) {
      showToast('Đã đạt giới hạn tối đa 10 ảnh');
      return;
    }

    const newThumb = document.createElement('div');
    newThumb.className = 'relative group rounded-lg overflow-hidden bg-surface-container aspect-square flex flex-col shadow-sm';
    newThumb.innerHTML = `
      <img src="placeholder" data-alt="Clear packaging snapshot of fresh product stock, barcode label preview, warehouse quality inspection check, high resolution clean lighting" class="w-full h-full object-cover" />
      <button type="button" onclick="this.parentElement.remove(); updatePhotoCount();" class="absolute top-1.5 right-1.5 w-6 h-6 rounded-full bg-inverse-surface/75 text-inverse-on-surface flex items-center justify-center hover:bg-error transition-colors">
        <span class="material-symbols-outlined text-[14px]">close</span>
      </button>
      <div class="absolute bottom-0 inset-x-0 bg-inverse-surface/60 py-0.5 text-center">
        <span class="font-label-sm text-label-sm text-surface-container-lowest">Ảnh chi tiết #${existing + 1}</span>
      </div>
    `;
    gallery.insertBefore(newThumb, gallery.lastElementChild);
    updatePhotoCount();
    showToast('Đã thêm 1 ảnh kiểm chứng mới');
  }

  function triggerSuccessToast() {
    showToast('Đã lưu thành công! Đang đồng bộ danh mục lưu kho...');
  }

<script>
  (function () {
    const searchInput = document.getElementById('facility-search-input');
    const searchClear = document.getElementById('facility-search-clear');
    const emptyState = document.getElementById('search-empty-state');
    const cardsContainer = document.getElementById('facilities-container');
    const resetBtn = document.getElementById('reset-search-btn');
    const filterChips = document.querySelectorAll('.filter-chip');

    let activeFilter = 'all';

    function filterCards() {
      const query = (searchInput.value || '').trim().toLowerCase();
      const cards = cardsContainer.querySelectorAll('.facility-card');
      let visibleCount = 0;

      cards.forEach(card => {
        const text = (card.getAttribute('data-name') || '') + ' ' + card.innerText.toLowerCase();
        let matchesQuery = !query || text.includes(query);
        let matchesTag = true;

        if (activeFilter === 'cool') {
          matchesTag = text.includes('kho mát') || text.includes('20°c') || text.includes('nhiệt ẩm');
        } else if (activeFilter === 'dock') {
          matchesTag = text.includes('cổng bốc dỡ') || text.includes('container') || text.includes('3.5t');
        } else if (activeFilter === 'available') {
          matchesTag = text.includes('khả dụng');
        }

        if (matchesQuery && matchesTag) {
          card.classList.remove('hidden');
          visibleCount++;
        } else {
          card.classList.add('hidden');
        }
      });

      if (visibleCount === 0) {
        emptyState.classList.remove('hidden');
        emptyState.classList.add('flex');
      } else {
        emptyState.classList.add('hidden');
        emptyState.classList.remove('flex');
      }

      if (query.length > 0) {
        searchClear.classList.remove('hidden');
      } else {
        searchClear.classList.add('hidden');
      }
    }

    searchInput.addEventListener('input', filterCards);

    searchClear.addEventListener('click', function () {
      searchInput.value = '';
      searchInput.focus();
      filterCards();
    });

    if (resetBtn) {
      resetBtn.addEventListener('click', function () {
        searchInput.value = '';
        activeFilter = 'all';
        filterChips.forEach(chip => {
          chip.classList.remove('bg-primary', 'text-on-primary');
          chip.classList.add('bg-surface-container-lowest', 'text-on-surface-variant');
        });
        const allChip = document.querySelector('[data-filter="all"]');
        if (allChip) {
          allChip.classList.add('bg-primary', 'text-on-primary');
          allChip.classList.remove('bg-surface-container-lowest', 'text-on-surface-variant');
        }
        filterCards();
      });
    }

    filterChips.forEach(chip => {
      chip.addEventListener('click', function () {
        filterChips.forEach(c => {
          c.classList.remove('bg-primary', 'text-on-primary');
          c.classList.add('bg-surface-container-lowest', 'text-on-surface-variant');
        });
        chip.classList.add('bg-primary', 'text-on-primary');
        chip.classList.remove('bg-surface-container-lowest', 'text-on-surface-variant');
        activeFilter = chip.getAttribute('data-filter') || 'all';
        filterCards();
      });
